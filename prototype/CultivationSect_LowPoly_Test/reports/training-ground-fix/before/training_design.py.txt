def torus_horizontal(name,c,r,minor,material,col,n=20):
    # Real collar/rope geometry with a small cross-section, not stacked cubes.
    verts=[]
    for i in range(n):
        a=i*math.tau/n
        for j in range(5):
            b=j*math.tau/5
            verts.append((c[0]+(r+minor*math.cos(b))*math.cos(a),c[1]+minor*math.sin(b),c[2]+(r+minor*math.cos(b))*math.sin(a)))
    faces=[(i*5+j,((i+1)%n)*5+j,((i+1)%n)*5+(j+1)%5,i*5+(j+1)%5) for i in range(n) for j in range(5)]
    geometry(name,verts,faces,material,col,True)

def boundary_run(axis,fixed,a,b):
    length=b-a
    at=((a+b)/2,.58,fixed) if axis=="x" else (fixed,.58,(a+b)/2)
    size=(length,.54,.50) if axis=="x" else (.50,.54,length)
    box("Low perimeter stone base",at,size,"Ashlar","02_Enclosure",True)
    cap=list(at);cap[1]=.87
    capsize=list(size);capsize[1]=.10;capsize[0 if axis=="z" else 2]=.62
    box("Perimeter stone coping",cap,capsize,"Coping","02_Enclosure")
    count=math.ceil(length/3.5)
    for i in range(count+1):
        p=a+length*i/count
        x,z=(p,fixed) if axis=="x" else (fixed,p)
        box("Fence mortised upright",(x,1.18,z),(.18,.68,.18),"Timber","02_Enclosure")
        box("Fence post cap",(x,1.55,z),(.23,.07,.23),"CarvedWood","02_Enclosure")
    for y in (1.02,1.38):
        pos=list(at);pos[1]=y
        dims=list(size);dims[1]=.11;dims[0 if axis=="z" else 2]=.13
        box("Timber perimeter rails",pos,dims,"CarvedWood","02_Enclosure")
    # Broad low slab also prevents players/cameras passing through tiny rails.
    blocker=list(size);blocker[1]=1.20
    pos=list(at);pos[1]=.91
    solids.append(dict(name="Perimeter collision",type="box",position=pos,size=blocker))

def pavilion(name,cx,cz,w,d,floor,demonstration=False):
    col="03_MainFrame"
    base_height=floor-.31
    box(name+" dressed stone dais",(cx,.31+base_height/2,cz),(w,base_height,d),"Ashlar","01_Foundation",True)
    box(name+" stone floor",(cx,floor-.025,cz),(w+.10,.05,d+.10),"Coping","01_Foundation")
    steps=round(base_height/.15)
    stair_w=7 if demonstration else 3
    for i in range(steps):
        height=(i+1)*base_height/steps
        z=cz+d/2+(steps-i)*.38-.19
        box(name+" stair tread",(cx,.31+height/2,z),(stair_w,height,.76),"Coping","01_Foundation")
    ramp(name+" smooth stair support",cx,stair_w,cz+d/2+steps*.38,cz+d/2,.31,floor+.002)
    column_y=floor+1.55
    eave=floor+3.3
    xs=np.linspace(cx-w/2+.65,cx+w/2-.65,5 if demonstration else 3)
    zs=[cz-d/2+.60,cz+d/2-.60]
    for z in zs:
        # Keep a real entrance bay above the stairs. Rear columns retain the
        # regular grid; the front center post is replaced with a wider opening.
        front_xs=[x for x in xs if abs(x-cx)>.01]
        if not demonstration: front_xs += [cx-1.95,cx+1.95]
        for x in (front_xs if z==zs[1] else xs):
            rod(name+" column stone foot",(x,floor,z),(x,floor+.20,z),.26,"Coping",col,12)
            rod(name+" structural column",(x,floor+.20,z),(x,eave-.20,z),.16,"Timber",col,12,True)
            collar(name+" column iron band",x,floor+.37,z,.164,col=col)

            for side in (-1,1):
                beam(name+" timber knee brace",(x,eave-.68,z),(x+side*.65,eave-.10,z),.14,.16,"CarvedWood",col)
    for z in zs:box(name+" front and rear tie beam",(cx,eave-.05,z),(w-.6,.25,.24),"Timber",col)
    for x in xs:box(name+" transverse bearer",(x,eave-.10,cz),(.20,.22,d-.55),"Timber",col)
    roof(name,cx,cz,w+1.4,d+1.5,eave,2.0 if demonstration else 1.35,.25)
    # Roof trusses stay below the surface and make the open shelter credible.
    ridge_y=eave+(2.0 if demonstration else 1.35)-.35

        for sign in (-1,1):beam(name+" principal rafter",(x,eave-.32,cz+sign*(d/2-.2)),(x,ridge_y,cz),.16,.20,"Timber","04_RoofStructure")

def export(items,filename,lod=False):
    deps=bpy.context.evaluated_depsgraph_get();grouped={}
    for original in items:
        data=bpy.data.meshes.new_from_object(original.evaluated_get(deps),preserve_all_data_layers=True,depsgraph=deps)
        obj=bpy.data.objects.new("Export "+original.name,data);scene.collection.objects.link(obj)
        grouped.setdefault(original.data.materials[0].name,[]).append(obj)
    joined=[]
    for name,parts in grouped.items():
        bpy.ops.object.select_all(action="DESELECT")
        for obj in parts:obj.select_set(True)
        bpy.context.view_layer.objects.active=parts[0]
        if len(parts)>1:bpy.ops.object.join()
        obj=bpy.context.object;obj.name=("LOD_" if lod else "Training_")+name.removeprefix("Training_")
        bm=bmesh.new();bm.from_mesh(obj.data)
        bmesh.ops.remove_doubles(bm,verts=list(bm.verts),dist=.000001)
        bmesh.ops.dissolve_degenerate(bm,edges=list(bm.edges),dist=.0000001)
        bm.to_mesh(obj.data);bm.free()
        if lod:
            mod=obj.modifiers.new("Distant simplification","DECIMATE");mod.ratio=.40
            bpy.ops.object.modifier_apply(modifier=mod.name)
            bm=bmesh.new();bm.from_mesh(obj.data)
            bmesh.ops.dissolve_degenerate(bm,edges=list(bm.edges),dist=.0000001)
            bm.to_mesh(obj.data);bm.free()
        obj.data.validate(verbose=False,clean_customdata=False);obj.data.update();obj.data.calc_loop_triangles();joined.append(obj)
    bpy.ops.object.select_all(action="DESELECT")
    for obj in joined:obj.select_set(True)
    pts=np.array([(v.co.x,v.co.z,-v.co.y) for o in joined for v in o.data.vertices])
    stats=dict(file=filename,triangles=sum(len(o.data.loop_triangles) for o in joined),material_batches=len(joined),bounds_min=pts.min(axis=0).tolist(),bounds_max=pts.max(axis=0).tolist())
    bpy.ops.export_scene.gltf(filepath=str(OUT/filename),export_format="GLB",use_selection=True,export_apply=True,export_yup=True,export_texcoords=True,export_normals=True,export_materials="EXPORT")
    for obj in joined:bpy.data.objects.remove(obj,do_unlink=True)
    return stats
